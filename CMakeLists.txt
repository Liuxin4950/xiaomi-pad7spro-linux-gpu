cmake_minimum_required(VERSION 3.22)
project(pad7spro_gpu VERSION 0.1.0 LANGUAGES C)
set(BRIDGE_DEPS_PREFIX "" CACHE PATH "Staged libhybris and Android headers")
if(NOT BRIDGE_DEPS_PREFIX)
  message(FATAL_ERROR "Set BRIDGE_DEPS_PREFIX to the staged dependency prefix")
endif()
find_package(Threads REQUIRED)
find_package(Vulkan REQUIRED)
find_library(HYBRIS_COMMON hybris-common PATHS "${BRIDGE_DEPS_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(HYBRIS_HARDWARE hardware PATHS "${BRIDGE_DEPS_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(sysvk SHARED src/sysvk.c)
target_include_directories(sysvk PRIVATE vendor/aosp "${BRIDGE_DEPS_PREFIX}/include/android")
target_link_libraries(sysvk PRIVATE "${HYBRIS_HARDWARE}" Threads::Threads Vulkan::Headers)
add_library(ahb-wrapper SHARED src/ahb-wrapper.c)
target_include_directories(ahb-wrapper PRIVATE "${BRIDGE_DEPS_PREFIX}/include")
target_link_libraries(ahb-wrapper PRIVATE "${HYBRIS_COMMON}" Threads::Threads)
set_target_properties(ahb-wrapper PROPERTIES VERSION 1.0.0 SOVERSION 1)
foreach(target sysvk ahb-wrapper)
  target_compile_features(${target} PRIVATE c_std_11)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
  target_link_options(${target} PRIVATE -Wl,-z,nodelete -Wl,-z,noexecstack -Wl,--no-undefined)
  set_target_properties(${target} PROPERTIES INSTALL_RPATH "${CMAKE_INSTALL_PREFIX}/lib")
endforeach()
add_executable(probe-vulkan src/probe-vulkan.c)
target_link_libraries(probe-vulkan PRIVATE Vulkan::Vulkan)
install(TARGETS sysvk ahb-wrapper LIBRARY DESTINATION lib)
install(TARGETS probe-vulkan RUNTIME DESTINATION bin)
